key DAX used:
HeadCount = COUNTROWS(Staff)

AvgLeaveBalance = AVERAGE(Staff[Leave Balance])

AvgSalary = AVERAGE(Staff[Salary])

MaxSalary = MAX(Staff[Salary])

MinSalary = MIN(Staff[Salary])

CumulativeHeadcount =
    VAR currentDate = LASTDATE(Staff[Date of Join])
RETURN
    CALCULATE(
        [Headcount],
        ALL(Staff[Date of Join]),
        Staff[Date of Join] <= currentDate
    )

LBL Over 20Days = CALCULATE(
    [HeadCount],
    Staff[Leave Balance] > 20)
